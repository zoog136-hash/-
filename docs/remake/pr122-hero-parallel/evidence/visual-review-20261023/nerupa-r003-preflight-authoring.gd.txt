func _json(path: String, data: Variant) -> void:
	var file := FileAccess.open(path, FileAccess.WRITE)
	assert(file != null)
	file.store_string(JSON.stringify(data, "  ") + "\n")

func _hash(image: Image) -> String:
	var hash := HashingContext.new()
	hash.start(HashingContext.HASH_SHA256)
	hash.update(image.get_data())
	return hash.finish().hex_encode()

func _initialize() -> void:
	call_deferred("_run")

func _part(name: String, polygon: Array, anchor: Vector2, kind: String, side: float) -> void:
	var uv := PackedVector2Array()
	for p: Array in polygon: uv.append(Vector2(p[0], p[1]))
	var triangles := Geometry2D.triangulate_polygon(uv)
	assert(not triangles.is_empty())
	if kind == "body":
		# Sample the curved surface inside the UV polygon, not just its outline.
		# Outline-only projection made the torso a nearly edge-on flat panel.
		var refined := PackedInt32Array()
		for i: int in range(0, triangles.size(), 3):
			var a: int = triangles[i]
			var b: int = triangles[i+1]
			var c: int = triangles[i+2]
			var center := uv.size()
			uv.append((uv[a]+uv[b]+uv[c])/3.0)
			refined.append_array(PackedInt32Array([a,b,center,b,c,center,c,a,center]))
		triangles = refined
	parts.append({"name": name, "uv": uv, "triangles": triangles, "anchor": anchor, "kind": kind, "side": side})

func _leg_bind(side: float) -> Array:
	return LEG_BIND.left if side < 0 else LEG_BIND.right

func _bind(uv: Vector2) -> Vector3:
	return Vector3(uv.x - 51.0, 0.0, 115.0 - uv.y)

func _surface_depth(x: float) -> float:
	return -sqrt(maxf(0.0, 12.0*12.0 - x*x)) * .6

func _leg(side: float, pose: Dictionary) -> Dictionary:
	var bind := _leg_bind(side)
	var hip_uv: Vector2 = bind[0]
	var knee_uv: Vector2 = bind[1]
	var foot_uv: Vector2 = bind[2]
	var hip := _bind(hip_uv) + Vector3(0,0,-pose.root_down)
	hip.y = _surface_depth(hip.x)
	var phase: float = pose.leg_phase + (0.0 if side < 0 else PI)
	var stride: float = pose.stride * cos(phase)
	var lift: float = pose.lift * maxf(0.0, sin(phase))
	var foot := _bind(foot_uv) + Vector3(0,stride,lift)
	# Both painted feet share one world ground; no image translation masquerades as a stride.
	foot.z = lift + float(pose.flight)
	var upper := hip_uv.distance_to(knee_uv)
	var lower := knee_uv.distance_to(foot_uv)
	# Keep the hip on the torso. Clamp only the forward stride if this splayed
	# portrait leg cannot reach it; moving the hip independently opened a seam.
	var reach_limit := upper + lower - .15
	var max_stride := sqrt(maxf(0.0, reach_limit*reach_limit - pow(foot.x-hip.x,2.0) - pow(foot.z-hip.z,2.0)))
	foot.y = hip.y + clampf(foot.y-hip.y,-max_stride,max_stride)
	var delta := foot - hip
	var reach := delta.length()
	var axis := delta / reach
	var a := (upper*upper - lower*lower + reach*reach) / (2.0*reach)
	var normal := Vector3(side,0,0)
	normal = (normal - axis * normal.dot(axis)).normalized()
	var knee := hip + axis*a + normal*sqrt(maxf(0.0,upper*upper-a*a))
	return {"hip":hip, "knee":knee, "foot":foot}

func _vertex(uv: Vector2, part: Dictionary, pose: Dictionary) -> Vector3:
	var point: Vector3
	var side: float = part.side
	if part.kind in ["upper_leg", "lower_leg"]:
		var leg := _leg(side,pose)
		var bind := _leg_bind(side)
		var a: Vector2 = bind[0]
		var b: Vector2 = bind[1]
		var c: Vector2 = bind[2]
		var start := a if part.kind == "upper_leg" else b
		var finish := b if part.kind == "upper_leg" else c
		var fraction := (uv-start).dot(finish-start) / start.distance_squared_to(finish)
		point = (leg.hip as Vector3).lerp(leg.knee, fraction) if part.kind == "upper_leg" else (leg.knee as Vector3).lerp(leg.foot, fraction)
	elif part.kind == "body":
		point = _bind(uv)
		# Curved volume avoids a flat cutout disappearing at side headings.
		point.y = _surface_depth(point.x)
		point.z -= pose.root_down
	else:
		var delta: Vector2 = uv - part.anchor
		var angle: float = pose.claw_swing * side if part.kind == "claw" else pose.antenna_sway * side
		delta = delta.rotated(angle)
		point = _bind(part.anchor + delta)
		point.y = _surface_depth(part.anchor.x-51.0) + (absf(delta.x) * .65 + maxf(0.0,delta.y) * .3 if part.kind == "claw" else -sin(delta.y / 35.0 * PI / 2.0) * 8.0)
		point.z -= pose.root_down
	point.y += pose.lunge
	if pose.is_death:
		point = point.rotated(Vector3.RIGHT, pose.fall)
		point.z = maxf(0.0, point.z)
	return point.rotated(Vector3.BACK, pose.yaw)

func _project(point: Vector3, height: float) -> Vector2:
	return Vector2(128.0 + point.x, 164.0 + point.y * .5 - point.z - height)

func _triangle(image: Image, uv: Array[Vector2], positions: Array[Vector2], shade: float) -> void:
	var scale := float(image.get_width()) / 256.0
	var a: Vector2 = positions[0] * scale
	var b: Vector2 = positions[1] * scale
	var c: Vector2 = positions[2] * scale
	var det: float = (b - a).cross(c - a)
	if absf(det) < .00001: return
	var bounds := Rect2(a, Vector2.ZERO).expand(b).expand(c).grow(1)
	var area := Rect2i(bounds).intersection(Rect2i(Vector2i.ZERO, image.get_size()))
	for y: int in range(area.position.y, area.end.y):
		for x: int in range(area.position.x, area.end.x):
			var relative := Vector2(x + .5, y + .5) - a
			var u := relative.cross(c - a) / det
			var v := (b - a).cross(relative) / det
			if u < -.0001 or v < -.0001 or u + v > 1.0001: continue
			var sample := Vector2i((uv[0] * (1.0 - u - v) + uv[1] * u + uv[2] * v).floor())
			if sample.x < 0 or sample.y < 0 or sample.x >= source.get_width() or sample.y >= source.get_height(): continue
			var color := source.get_pixelv(sample)
			# Portrait-only outer black matte is excluded from derived part layers.
			if color.a <= 0.0 or maxf(color.r, maxf(color.g, color.b)) <= .10: continue
			color.r *= shade
			color.g *= shade
			color.b *= shade
			var old := image.get_pixel(x, y)
			var alpha: float = color.a + old.a * (1.0 - color.a)
			image.set_pixel(x, y, Color((color.r * color.a + old.r * old.a * (1.0 - color.a)) / alpha, (color.g * color.a + old.g * old.a * (1.0 - color.a)) / alpha, (color.b * color.a + old.b * old.a * (1.0 - color.a)) / alpha, alpha))

func _pose(role: String, direction: int, phase: int) -> Dictionary:
	var pose := {"yaw": (direction-2)*PI/4.0, "leg_phase":phase*TAU/6.0, "stride":0.0, "lift":0.0, "root_down":0.0, "flight":0.0, "height":0.0, "claw_swing":0.0, "antenna_sway":0.0, "lunge":0.0, "fall":0.0, "shade":1.0, "body_shade":.85 + .15*cos((direction-2)*PI/4.0), "is_death":role=="death"}
	match role:
		"idle":
			pose.antenna_sway = [.0,.035,-.025][phase]
			pose.claw_swing = [.0,.015,-.02][phase]
		"run":
			pose.stride = 7.0
			pose.lift = 5.0
			pose.root_down = [2.0,3.0,-1.0,2.0,3.0,-1.0][phase]
			pose.flight = 2.0 if phase % 3 == 2 else 0.0
			pose.claw_swing = .1*sin(pose.leg_phase)
			pose.antenna_sway = .07*cos(pose.leg_phase-.4)
		"attack":
			pose.claw_swing = [.18,-.65,-.4,-.1,.04,.0][phase]
			pose.lunge = [0.0,-9.0,-5.0,-2.0,1.0,0.0][phase]
			pose.antenna_sway = [.03,-.1,-.04,.02,.03,.0][phase]
		"hit":
			pose.lunge = [3.0,6.0,2.0][phase]
			pose.claw_swing = [.1,.2,.05][phase]
		"death":
			pose.fall = [.0,.25,.6,1.0,1.4,PI/2.0][phase]
			pose.claw_swing = [.0,.12,.3,.5,.6,.6][phase]
			pose.antenna_sway = [.0,.0,.05,.1,.15,.15][phase]
			pose.shade = [1.0,1.0,.98,.95,.9,.9][phase]
	return pose

func _render(pose: Dictionary) -> Image:
	var image := Image.create(1024, 1024, false, Image.FORMAT_RGBA8)
	var drawing: Array = []
	for part: Dictionary in parts:
		var points: Array[Vector2] = []
		var depth := 0.0
		for uv: Vector2 in part.uv:
			var transverse := 0.0
			var ribbon_normal := Vector2.ZERO
			var sample := uv
			if part.kind in ["upper_leg", "lower_leg"]:
				var bind := _leg_bind(part.side)
				var a: Vector2 = bind[0]
				var b: Vector2 = bind[1]
				var c: Vector2 = bind[2]
				var start := a if part.kind == "upper_leg" else b
				var finish := b if part.kind == "upper_leg" else c
				var axis := (finish-start).normalized()
				transverse = axis.cross(uv-start)
				sample = start + axis*axis.dot(uv-start)
				var screen_axis := (_project(_vertex(finish,part,pose),pose.height) - _project(_vertex(start,part,pose),pose.height)).normalized()
				ribbon_normal = Vector2(-screen_axis.y,screen_axis.x)
			var point := _vertex(sample, part, pose)
			var projected := _project(point, pose.height)
			projected += ribbon_normal * transverse * .9 # Perpendicular painted width follows the projected bone.
			points.append(projected)
			depth += point.y
		drawing.append({"part": part, "points": points, "depth": depth / part.uv.size()})
	drawing.sort_custom(func(a: Dictionary, b: Dictionary) -> bool: return a.depth < b.depth)
	for item: Dictionary in drawing:
		var part: Dictionary = item.part
		for index: int in range(0, part.triangles.size(), 3):
			var uv: Array[Vector2] = []
			var points: Array[Vector2] = []
			for offset: int in range(3):
				var vertex: int = part.triangles[index + offset]
				uv.append(part.uv[vertex])
				points.append(item.points[vertex])
			_triangle(image, uv, points, pose.shade * (pose.body_shade if part.kind == "body" else 1.0))
	image.generate_mipmaps()
	image.resize(256, 256, Image.INTERPOLATE_TRILINEAR)
	image.clear_mipmaps()
	return image

func _part_preflight() -> void:
	var key := FileAccess.get_sha256(SOURCE) + ":" + FileAccess.get_sha256("res://docs/remake/pr122-hero-parallel/bake_fire_nerupa.gd")
	var receipt := out + "part-preflight.json"
	if FileAccess.file_exists(receipt):
		var previous: Dictionary = JSON.parse_string(FileAccess.get_file_as_string(receipt))
		assert(previous.cache_key == key and FileAccess.get_sha256(out+"part-preflight.png") == previous.png_sha256)
		print("HERO_NERUPA_PART_PREFLIGHT_REUSED poses=8 writes=0")
		return
	var sheet := Image.create(512,1024,false,Image.FORMAT_RGBA8)
	var rows: Array = []
	var front_width := 0
	var side_width := 0
	for direction: int in [2,0,4,6]:
		for phase: int in [0,2]:
			var pose := _pose("run",direction,phase)
			var image := _render(pose)
			var box := image.get_used_rect()
			assert(box.position.x > 0 and box.position.y > 0 and box.end.x < 256 and box.end.y < 256)
			if direction == 2: front_width = maxi(front_width,box.size.x)
			if direction in [0,4]: side_width = box.size.x if side_width == 0 else mini(side_width,box.size.x)
			for side: float in [-1.0,1.0]:
				var bind := _leg_bind(side)
				var leg := _leg(side,pose)
				assert(absf((leg.hip as Vector3).distance_to(leg.knee)-(bind[0] as Vector2).distance_to(bind[1])) < .02)
				assert(absf((leg.knee as Vector3).distance_to(leg.foot)-(bind[1] as Vector2).distance_to(bind[2])) < .02)
				assert(leg.foot.z >= 0.0)
				var bind_hip := _bind(bind[0]) + Vector3(0,0,-pose.root_down)
				assert(absf(leg.hip.z-bind_hip.z) < .001)
			var index := rows.size()
			sheet.blit_rect(image,Rect2i(0,0,256,256),Vector2i((index%2)*256,(index/2)*256))
			rows.append({"direction":direction,"phase":phase,"rgba_sha256":_hash(image),"bbox":[box.position.x,box.position.y,box.size.x,box.size.y]})
	assert(side_width >= 20 and float(side_width)/front_width >= .28)
	assert(sheet.save_png(out+"part-preflight.png") == OK)
	_json(receipt,{"cache_key":key,"status":"PASS_BIND_LENGTHS_FIXED_HIPS_NO_EDGE_CLIPPING_SIDE_VOLUME","poses":rows,"front_width":front_width,"minimum_side_width":side_width,"png_sha256":FileAccess.get_sha256(out+"part-preflight.png"),"final_approved":false})
	print("HERO_NERUPA_PART_PREFLIGHT_PASS poses=8 front_width=%d side_width=%d" % [front_width,side_width])
